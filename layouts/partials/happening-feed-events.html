{{- /* Events from data/happening/*.yaml in the page's language, sorted by start, for the JSON and iCalendar feeds and the homepage's "New and pinned" panel. Takes (dict "topic" "careers") to keep one topic, or (dict) for all. Dates are as written in the data (Hobart time with its offset), plus UTC and date-only forms for iCalendar. */ -}} {{- $lang := site.Language.Lang -}} {{- $topic := .topic | default "" -}} {{- $happening := site.GetPage "/happening" -}} {{- $events := slice -}} {{- range $id, $e := hugo.Data.happening -}} {{- if or (not $topic) (in ($e.topics | default slice) $topic) -}} {{- $start := time $e.start -}} {{- $end := time ($e.end | default $e.start) -}} {{- $image := "" -}} {{- with $e.image }}{{ $image = absURL (strings.TrimPrefix "/" .) }}{{ end -}} {{- $page := "" -}} {{- with $happening }}{{ $page = printf "%s#event-%s" .Permalink $id }}{{ end -}} {{- $events = $events | append (dict "id" $id "title" (partial "happening-text.html" (dict "v" $e.title "lang" $lang)) "start" ($start.Format "2006-01-02T15:04:05Z07:00") "end" ($end.Format "2006-01-02T15:04:05Z07:00") "start_utc" ($start.UTC.Format "20060102T150405Z") "end_utc" ($end.UTC.Format "20060102T150405Z") "start_day" ($start.Format "20060102") "end_day_after" (($end.AddDate 0 0 1).Format "20060102") "all_day" ($e.all_day | default false) "date_tbc" ($e.date_tbc | default false) "type" ($e.type | default "event") "format" ($e.format | default "in_person") "topics" ($e.topics | default slice) "host" ($e.host | default "") "location" (partial "happening-text.html" (dict "v" $e.location "lang" $lang)) "cost" (partial "happening-text.html" (dict "v" $e.cost "lang" $lang)) "summary" (partial "happening-text.html" (dict "v" $e.summary "lang" $lang)) "url" ($e.url | default "") "page" $page "image" $image "pinned" ($e.pinned | default 0) "added" ($e.added | default "") "ts" $start.Unix ) -}} {{- end -}} {{- end -}} {{- return (sort $events "ts") -}}
